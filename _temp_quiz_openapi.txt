openapi: 3.1.0
info:
  title: Quiz Sessions API
  description: |
    API for managing quiz sessions across all quiz types (declensions, conjugations, vocabulary).

    Based on actual code at `services/quiz-service` — Java 21 + WebFlux (Reactor) + R2DBC.

    Manages the full lifecycle: start/resume, submitting answers, completion, retake, new-quiz.
    Distractors are generated on-the-fly (generate-on-read) and are NOT persisted.
    Questions are stored in quiz-service database (session_questions table) — the ONLY
    persistent storage of session questions in the entire system.
    Events (QuizAnsweredEvent, QuizSessionStatusChangedEvent) are published via Outbox Pattern.
  version: 1.0.0

servers:
  - url: http://localhost:8082/api/v1
    description: Local development (direct to quiz-service)
  - url: http://localhost:8090/api/v1
    description: Local development (via API Gateway)

tags:
  - name: quiz-sessions
    description: Quiz session lifecycle management

paths:

  # ────────── START / START-OR-RESUME / RESUME ──────────

  /quiz/{slug}/sessions/start:
    post:
      tags: [quiz-sessions]
      summary: Start a new quiz session (or resume if in progress)
      description: |
        Starts a new quiz session for the given lesson. If an in-progress session
        already exists for the same user and lesson, it resumes that session instead.

        Calls content-service POST /api/v1/content/lessons/{id}/generate-quiz-data
        to generate questions, saves them to session_questions table, and returns
        the session with generated distractors.
      operationId: startSession
      parameters:
        - $ref: 'parameters.yaml#/SlugParam'
        - $ref: 'parameters.yaml#/LessonIdParam'
        - $ref: 'parameters.yaml#/XUserIdHeader'
        - $ref: 'parameters.yaml#/XUserLocaleHeader'
      responses:
        '200':
          description: Session started or resumed successfully
          content:
            application/json:
              schema:
                $ref: 'schemas/session.yaml#/StartOrResumeResponse'
        '404':
          description: Quiz not found
          content:
            application/json:
              schema:
                $ref: 'schemas/common.yaml#/ErrorResponse'

  /quiz/{slug}/sessions/start-or-resume:
    post:
      tags: [quiz-sessions]
      summary: Start a new quiz session or resume the latest in-progress session
      description: |
        Explicitly starts a new session or resumes the latest in-progress session
        for the given lesson and user. Same logic as /start but with explicit intent.
      operationId: startOrResumeSession
      parameters:
        - $ref: 'parameters.yaml#/SlugParam'
        - $ref: 'parameters.yaml#/LessonIdParam'
        - $ref: 'parameters.yaml#/XUserIdHeader'
        - $ref: 'parameters.yaml#/XUserLocaleHeader'
      responses:
        '200':
          description: Session started or resumed successfully
          content:
            application/json:
              schema:
                $ref: 'schemas/session.yaml#/StartOrResumeResponse'
        '404':
          description: Quiz not found
          content:
            application/json:
              schema:
                $ref: 'schemas/common.yaml#/ErrorResponse'

  /quiz/{slug}/sessions/{sessionId}/resume:
    get:
      tags: [quiz-sessions]
      summary: Resume an existing quiz session by session ID
      description: |
        Resumes a specific quiz session identified by sessionId.
        Loads questions from quiz.session_questions (not from content-service),
        regenerates distractors on-the-fly via DeclensionOptionGeneratorService
        or LexicalOptionGeneratorService.
      operationId: resumeSession
      parameters:
        - $ref: 'parameters.yaml#/SlugParam'
        - $ref: 'parameters.yaml#/SessionIdParam'
        - $ref: 'parameters.yaml#/XUserIdHeader'
        - $ref: 'parameters.yaml#/XUserLocaleHeader'
      responses:
        '200':
          description: Session resumed successfully
          content:
            application/json:
              schema:
                $ref: 'schemas/session.yaml#/StartOrResumeResponse'
        '404':
          description: Session not found
          content:
            application/json:
              schema:
                $ref: 'schemas/common.yaml#/ErrorResponse'

  # ────────── ANSWER ──────────

  /quiz/{slug}/sessions/{sessionId}/answer:
    post:
      tags: [quiz-sessions]
      summary: Submit an answer for a question in a quiz session
      description: |
        Submits a user's answer to a specific question.
        - Checks if the question was already answered (returns 409 Conflict)
        - Compares the answer by string value (IAST or translation), NOT by option ID
          (since distractors are generated on-the-fly and have ephemeral IDs)
        - Saves QuizAnswer, updates QuizSession (answered_questions, score)
        - Updates word_score or grammar_form_score via ScoreUpdateStrategy
        - Publishes QuizAnsweredEvent via Outbox Pattern
      operationId: submitAnswer
      parameters:
        - $ref: 'parameters.yaml#/SlugParam'
        - $ref: 'parameters.yaml#/SessionIdParam'
        - $ref: 'parameters.yaml#/XUserIdHeader'
        - $ref: 'parameters.yaml#/XUserLocaleHeader'
      requestBody:
        required: true
        content:
          application/json:
            schema:
              $ref: 'schemas/answers.yaml#/AnswerRequest'
      responses:
        '200':
          description: Answer submitted successfully
          content:
            application/json:
              schema:
                $ref: 'schemas/answers.yaml#/AnswerResponse'
        '404':
          description: Session or question not found
          content:
            application/json:
              schema:
                $ref: 'schemas/common.yaml#/ErrorResponse'
        '409':
          description: Question already answered
          content:
            application/json:
              schema:
                $ref: 'schemas/common.yaml#/ErrorResponse'

  # ────────── COMPLETE ──────────

  /quiz/{slug}/sessions/{sessionId}/complete:
    post:

        Marks the session as COMPLETED, sets completedAt timestamp,

        Publishes QuizSessionStatusChangedEvent (status=COMPLETED) via Outbox Pattern.
      operationId: completeSession
      parameters:
        - $ref: 'parameters.yaml#/SlugParam'
        - $ref: 'parameters.yaml#/SessionIdParam'
        - $ref: 'parameters.yaml#/XUserIdHeader'
      responses:
        '200':
          description: Session completed successfully
          content:
            application/json:
              schema:
                $ref: 'schemas/answers.yaml#/CompleteSessionResponse'
        '404':
          description: Session not found
          content:
            application/json:
              schema:
                $ref: 'schemas/common.yaml#/ErrorResponse'

  # ────────── RETAKE ──────────

  /quiz/{slug}/sessions/{sessionId}/retake:
    post:
      tags: [quiz-sessions]
      summary: Retake an existing quiz session, clearing answers and resetting progress
      description: |
        Resets the session: deletes all QuizAnswer records, resets score and
        answered_questions to 0, sets status back to IN_PROGRESS, updates startedAt.
        Publishes QuizSessionStatusChangedEvent (status=IN_PROGRESS) via Outbox Pattern.
      operationId: retakeSession
      parameters:
        - $ref: 'parameters.yaml#/SlugParam'
        - $ref: 'parameters.yaml#/SessionIdParam'
        - $ref: 'parameters.yaml#/XUserIdHeader'
        - $ref: 'parameters.yaml#/XUserLocaleHeader'
      responses:
        '200':
          description: Session reset for retake
          content:
            application/json:
              schema:
                $ref: 'schemas/session.yaml#/StartOrResumeResponse'
        '404':
          description: Session not found
          content:
            application/json:
              schema:
                $ref: 'schemas/common.yaml#/ErrorResponse'

  # ────────── NEW QUIZ ──────────

  /quiz/{slug}/sessions/{sessionId}/new-quiz:
    post:
      tags: [quiz-sessions]
      summary: Complete current session and start a new quiz session of the same type
      description: |
        Completes the current session (marks as COMPLETED), then generates and
        starts a completely new quiz session for the same lesson.
      operationId: startNewQuizFromExistingSession
      parameters:
        - $ref: 'parameters.yaml#/SlugParam'
        - $ref: 'parameters.yaml#/SessionIdParam'
        - $ref: 'parameters.yaml#/XUserIdHeader'
        - $ref: 'parameters.yaml#/XUserLocaleHeader'
      responses:
        '200':
          description: New quiz session started
          content:
            application/json:
              schema:
                $ref: 'schemas/session.yaml#/StartOrResumeResponse'
        '404':
          description: Session not found
          content:
            application/json:
              schema:
                $ref: 'schemas/common.yaml#/ErrorResponse'
